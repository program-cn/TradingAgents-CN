from contextlib import asynccontextmanager

@asynccontextmanager
async def lifespan(app: FastAPI):
    """应用生命周期管理"""
    # 启动时：初始化数据库连接、加载配置、启动定时任务
    await startup()
    yield

    await shutdown()

app = FastAPI(
    title="TradingAgents-CN API",
    lifespan=lifespan

#### 3.1.3 多智能体工作流 (tradingagents/graph/trading_graph.py)

```python
# 文件位置：tradingagents/graph/trading_graph.py
# 代码行数：1398 行
# 核心职责：编排多智能体协作流程

from langgraph.graph import StateGraph

class TradingAgentsGraph:
    """多智能体交易分析工作流"""
    
    def __init__(self):
        self.graph = self._build_graph()
    
    def _build_graph(self) -> StateGraph:
        """构建工作流图"""
        workflow = StateGraph(AgentState)
        
        # 添加节点（各智能体）
        workflow.add_node("market_analyst", self.market_analyst)
        workflow.add_node("fundamentals_analyst", self.fundamentals_analyst)
        workflow.add_node("news_analyst", self.news_analyst)
        workflow.add_node("research_team", self.research_team)
        workflow.add_node("risk_manager", self.risk_manager)
        workflow.add_node("trader", self.trader)
        
        # 定义边（工作流）
        workflow.set_entry_point("market_analyst")
        workflow.add_edge("market_analyst", "fundamentals_analyst")
        workflow.add_edge("fundamentals_analyst", "news_analyst")
        workflow.add_edge("news_analyst", "research_team")
        workflow.add_edge("research_team", "risk_manager")
        workflow.add_edge("risk_manager", "trader")
        
        return workflow.compile()
    
    async def run(self, symbol: str) -> dict:
        """执行分析工作流"""
        return await self.graph.ainvoke({"symbol": symbol})
```
